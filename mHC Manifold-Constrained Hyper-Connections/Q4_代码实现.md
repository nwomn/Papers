## 2. 核心算法实现

### 2.1 Sinkhorn-Knopp 算法

将任意矩阵投影到双随机矩阵流形：

$$\mathbf{M}^{(t)} = \mathcal{T}_r\left( \mathcal{T}_c(\mathbf{M}^{(t-1)}) \right)$$

```python
def sinkhorn_knopp(M: torch.Tensor, num_iters: int = 20) -> torch.Tensor:
    """
    将矩阵投影到双随机矩阵流形 (Birkhoff Polytope)

    Args:
        M: 输入矩阵 [batch, n, n]
        num_iters: 迭代次数 (论文使用 t_max = 20)

    Returns:
        双随机矩阵 [batch, n, n]，满足：
        - 每行之和 = 1
        - 每列之和 = 1
        - 所有元素 >= 0
    """
    # Step 1: 取指数保证非负
    M = torch.exp(M)

    # Step 2: 交替行/列归一化
    for _ in range(num_iters):
        # 行归一化: 每行之和 = 1
        M = M / M.sum(dim=-1, keepdim=True)
        # 列归一化: 每列之和 = 1
        M = M / M.sum(dim=-2, keepdim=True)

    return M
```

### 2.2 mHC 层完整实现

```python
import torch
import torch.nn as nn
import torch.nn.functional as F


class ManifoldConstrainedHyperConnections(nn.Module):
    """
    Manifold-Constrained Hyper-Connections (mHC)

    论文公式:
        x_{l+1} = P_{M^res}(H_l^res) * x_l + H_l^post^T * F(H_l^pre * x_l, W_l)

    Args:
        dim: 特征维度 C
        n_streams: 通道数 n (默认 4)
        alpha_init: 门控因子初始值 (默认 0.01)
        sinkhorn_iters: Sinkhorn-Knopp 迭代次数 (默认 20)
    """

    def __init__(
        self,
        dim: int,
        n_streams: int = 4,
        alpha_init: float = 0.01,
        sinkhorn_iters: int = 20
    ):
        super().__init__()
        self.dim = dim
        self.n = n_streams
        self.sinkhorn_iters = sinkhorn_iters

        # 动态映射的线性投影
        # 输入: flatten 后的 x_l, 维度 n*C
        # 输出: pre (n), post (n), res (n^2)
        self.proj = nn.Linear(n_streams * dim, n_streams * n_streams + 2 * n_streams, bias=False)

        # 静态偏置
        self.b_pre = nn.Parameter(torch.zeros(1, n_streams))
        self.b_post = nn.Parameter(torch.zeros(1, n_streams))
        self.b_res = nn.Parameter(torch.zeros(n_streams, n_streams))

        # 门控因子 (初始化为小值，使得初始状态接近恒等映射)
        self.alpha_pre = nn.Parameter(torch.tensor(alpha_init))
        self.alpha_post = nn.Parameter(torch.tensor(alpha_init))
        self.alpha_res = nn.Parameter(torch.tensor(alpha_init))

        # RMSNorm 用于归一化
        self.norm = RMSNorm(n_streams * dim)

    def forward(
        self,
        x: torch.Tensor,           # [batch, n, dim]
        layer_fn: callable         # 层函数 F(·, W_l)
    ) -> torch.Tensor:
        """
        Args:
            x: 输入隐藏状态 [batch, n_streams, dim]
            layer_fn: 层计算函数 (Attention 或 FFN)

        Returns:
            更新后的隐藏状态 [batch, n_streams, dim]
        """
        batch_size = x.shape[0]

        # ========== Step 1: 计算映射矩阵 ==========
        # Flatten 并归一化
        x_flat = x.view(batch_size, -1)  # [batch, n*dim]
        x_norm = self.norm(x_flat)       # [batch, n*dim]

        # 线性投影
        proj_out = self.proj(x_norm)     # [batch, n^2 + 2n]

        # 分割输出
        h_pre = proj_out[:, :self.n]                           # [batch, n]
        h_post = proj_out[:, self.n:2*self.n]                  # [batch, n]
        h_res = proj_out[:, 2*self.n:].view(batch_size, self.n, self.n)  # [batch, n, n]

        # 应用门控和偏置
        H_pre_raw = self.alpha_pre * h_pre + self.b_pre        # [batch, n]
        H_post_raw = self.alpha_post * h_post + self.b_post    # [batch, n]
        H_res_raw = self.alpha_res * h_res + self.b_res        # [batch, n, n]

        # ========== Step 2: 流形约束 ==========
        # H_pre: Sigmoid 保证非负
        H_pre = torch.sigmoid(H_pre_raw)           # [batch, n], 范围 (0, 1)

        # H_post: 2 * Sigmoid 保证非负且范围更大
        H_post = 2 * torch.sigmoid(H_post_raw)     # [batch, n], 范围 (0, 2)

        # H_res: Sinkhorn-Knopp 投影到双随机矩阵
        H_res = self.sinkhorn_knopp(H_res_raw)     # [batch, n, n]

        # ========== Step 3: 应用映射 ==========
        # Pre-mapping: 汇聚 n 通道到 1 通道作为层输入
        # h_in = sum_j H_pre_j * x_j
        h_in = torch.einsum('bn,bnd->bd', H_pre, x)  # [batch, dim]

        # 层计算
        h_out = layer_fn(h_in)  # [batch, dim]

        # Post-mapping: 分发层输出到 n 通道
        # h_post_out_i = H_post_i * h_out
        h_post_out = H_post.unsqueeze(-1) * h_out.unsqueeze(1)  # [batch, n, dim]

        # Residual mapping: 通道间混合
        # x_res_i = sum_j H_res_ij * x_j
        x_res = torch.einsum('bij,bjd->bid', H_res, x)  # [batch, n, dim]

        # ========== Step 4: 残差合并 ==========
        x_new = x_res + h_post_out  # [batch, n, dim]

        return x_new

    def sinkhorn_knopp(self, M: torch.Tensor) -> torch.Tensor:
        """Sinkhorn-Knopp 投影算法"""
        M = torch.exp(M)
        for _ in range(self.sinkhorn_iters):
            M = M / M.sum(dim=-1, keepdim=True)  # 行归一化
            M = M / M.sum(dim=-2, keepdim=True)  # 列归一化
        return M


class RMSNorm(nn.Module):
    """Root Mean Square Layer Normalization"""

    def __init__(self, dim: int, eps: float = 1e-20):
        super().__init__()
        self.eps = eps
        self.weight = nn.Parameter(torch.ones(dim))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        rms = torch.sqrt(torch.mean(x ** 2, dim=-1, keepdim=True) + self.eps)
        return x / rms * self.weight
```

## 3. 在 Transformer 中使用 mHC

### 3.1 修改 Transformer Block

```python
class TransformerBlockWithMHC(nn.Module):
    """使用 mHC 的 Transformer Block"""

    def __init__(self, dim: int, n_heads: int, n_streams: int = 4):
        super().__init__()
        self.dim = dim
        self.n_streams = n_streams

        # mHC 层
        self.mhc_attn = ManifoldConstrainedHyperConnections(dim, n_streams)
        self.mhc_ffn = ManifoldConstrainedHyperConnections(dim, n_streams)

        # 标准 Transformer 组件
        self.attn = MultiHeadAttention(dim, n_heads)
        self.ffn = FeedForward(dim)
        self.norm1 = RMSNorm(dim)
        self.norm2 = RMSNorm(dim)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: [batch, n_streams, dim]
        Returns:
            [batch, n_streams, dim]
        """
        # Attention with mHC
        def attn_fn(h):
            return self.attn(self.norm1(h))
        x = self.mhc_attn(x, attn_fn)

        # FFN with mHC
        def ffn_fn(h):
            return self.ffn(self.norm2(h))
        x = self.mhc_ffn(x, ffn_fn)

        return x


class TransformerWithMHC(nn.Module):
    """完整的 mHC Transformer"""

    def __init__(
        self,
        vocab_size: int,
        dim: int,
        n_layers: int,
        n_heads: int,
        n_streams: int = 4
    ):
        super().__init__()
        self.n_streams = n_streams

        # Embedding
        self.embed = nn.Embedding(vocab_size, dim)

        # 扩展到 n_streams (初始时复制到所有通道)
        self.expand = nn.Linear(dim, n_streams * dim)

        # Transformer blocks
        self.blocks = nn.ModuleList([
            TransformerBlockWithMHC(dim, n_heads, n_streams)
            for _ in range(n_layers)
        ])

        # 收缩回单通道
        self.contract = nn.Linear(n_streams * dim, dim)

        # Output head
        self.norm = RMSNorm(dim)
        self.head = nn.Linear(dim, vocab_size)

    def forward(self, tokens: torch.Tensor) -> torch.Tensor:
        batch_size, seq_len = tokens.shape

        # Embedding
        x = self.embed(tokens)  # [batch, seq, dim]

        # 扩展到 n_streams
        x = self.expand(x)  # [batch, seq, n*dim]
        x = x.view(batch_size, seq_len, self.n_streams, -1)  # [batch, seq, n, dim]
        x = x.transpose(1, 2)  # [batch, n, seq, dim]

        # 这里简化处理，实际需要更复杂的 reshape
        # ...

        # Transformer blocks
        for block in self.blocks:
            x = block(x)

        # 收缩并输出
        x = x.transpose(1, 2).reshape(batch_size, seq_len, -1)
        x = self.contract(x)
        x = self.norm(x)
        logits = self.head(x)

        return logits
```
